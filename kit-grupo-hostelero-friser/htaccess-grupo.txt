# Redirecciones de los locales de Grupo Hostelero Friser.
# Pega estas líneas en el .htaccess de cada web del grupo, justo después de "RewriteEngine On"
# y antes de la regla que fuerza HTTPS y el dominio sin www.
# Cuando un local estrene web, cambia solo su línea por su dominio en las cuatro webs y actualiza locales.json.
# Si su web tiene versión inglesa, añade también su línea /ir/en/<slug>: las páginas en inglés enlazan ahí.
# Si una web es la de uno de los locales, su propia línea /ir/ puede quedarse: no se usa en su bloque.

RewriteRule ^ir/la-bodeguita/?$ https://labodeguitadelaesquina.com/ [R=302,L]
RewriteRule ^ir/en/la-bodeguita/?$ https://labodeguitadelaesquina.com/en/ [R=302,L]
RewriteRule ^ir/pub-colonial/?$ https://pubcolonial.com/ [R=302,L]
RewriteRule ^ir/en/pub-colonial/?$ https://pubcolonial.com/en/ [R=302,L]
RewriteRule ^ir/taberna-asturiana/?$ https://tabernaasturiana.com/ [R=302,L]
RewriteRule ^ir/en/taberna-asturiana/?$ https://tabernaasturiana.com/en/ [R=302,L]
RewriteRule ^ir/excalibur/?$ https://excaliburirishpub.com/ [R=302,L]
RewriteRule ^ir/en/excalibur/?$ https://excaliburirishpub.com/en/ [R=302,L]
